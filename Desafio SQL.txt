CREATE SCHEMA desafio;

USE desafio;

CREATE TABLE regioes(
 id_regiao INT(10),
 nome VARCHAR(90) NOT NULL,
 estado VARCHAR(40) NOT NULL,
 potencial VARCHAR(50) NOT NULL,
 
 PRIMARY KEY (id_regiao)
);

CREATE TABLE usinas(
 id_usina INT(10),
 nome VARCHAR(100) NOT NULL,
 cnpj VARCHAR(14) NOT NULL,
 capacidade_producao FLOAT NOT NULL,
 id_regiao INT(10),
 
 PRIMARY KEY (id_usina),
 FOREIGN KEY (id_regiao)
  REFERENCES regioes(id_regiao)
);

CREATE TABLE geracao(
 id_geracao INT(10),
 qtd_gerada FLOAT NOT NULL,
 horas_operadas FLOAT NULL,
 id_usina INT(10),

 PRIMARY KEY (id_geracao),
 FOREIGN KEY (id_usina)
  REFERENCES usinas(id_usina)
);
